class Product:
    def __init__(self, name, price):
        self.name = name
        self.price = price

    @classmethod
    def from_csv(cls, line):
        name, price = line.split(",")
        return cls(name, int(price))

p = Product.from_csv("mouse,45000")

해설 보기
def validate_score(score):
    if not 0 <= score <= 100:
        raise ValueError("점수 범위 오류")
    return score

assert validate_score(0) == 0
assert validate_score(100) == 100
for bad in [-1, 101]:
    try:
        validate_score(bad)
    except ValueError:
        pass
    else:
        raise AssertionError("ValueError가 필요합니다")
